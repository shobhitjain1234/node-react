// Node.js Express Setup with MVC, Knex, Joi, dotenv - Beginner Notes

// --- Overview ---

1. app.js is the main entry file of the project. This is where the server starts and routes are defined using app.use(). It connects all routes to the app.
2. userRoute.js contains all API endpoints, e.g., GET /users, POST /users. It maps the API route to the corresponding controller function.
3. controller files contain the logic for each route. They handle requests, responses, and business logic. They use models to interact with the database.
4. model files contain database queries written using Knex. These files handle data fetching, insertion, update, delete, etc.
5. db.js file sets up the Knex configuration and initializes the database connection.

// --- Middleware ---

1. Middleware functions run before controller logic. They accept three arguments: req, res, and next.
2. They are useful for validation, authentication, logging, etc.

// Example:
function exampleMiddleware(req, res, next) {
  console.log("Request received at:", new Date());
  next(); // call next function
}

// You can also use middleware to validate inputs.
function validateName(req, res, next) {
  if (!req.body.name) {
    return res.status(400).json({ error: "Name is required" });
  }
  next();
}

// --- Joi for Validation ---

 Joi is a library used to validate incoming data such as form inputs or API request bodies.

// Install:
npm install joi

// Example usage:
const Joi = require("joi");

const schema = Joi.object({
  name: Joi.string().required(),
  email: Joi.string().email().required()
});

const result = schema.validate(req.body);
if (result.error) {
  return res.status(400).json({ error: result.error.details[0].message });
}

// --- Knex Setup and Migration ---

// 1. Install Knex and MySQL2:
npm install knex mysql2

// 2. Initialize Knex:
npx knex init

// 3. Configure knexfile.js:
module.exports = {
  development: {
    client: "mysql2",
    connection: {
      host: "127.0.0.1",
      user: "your_mysql_user",
      password: "your_mysql_password",
      database: "your_database_name",
    },
    migrations: {
      directory: "./migrations",
    },
  },
};

// 4. Create a migration file:
npx knex migrate:make create_company_table

// 5. Edit the migration file:
exports.up = function (knex) {
  return knex.schema.createTable("company", (table) => {
    table.increments("id").primary();
    table.string("name").notNullable();
    table.string("email").unique().notNullable();
    table.string("address");
    table.timestamps(true, true); // created_at and updated_at
  });
};

exports.down = function (knex) {
  return knex.schema.dropTable("company");
};

// 6. Run the migration:
npx knex migrate:latest

// 7. Rollback last migration:
npx knex migrate:rollback
